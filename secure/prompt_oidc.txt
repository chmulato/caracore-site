- Autenticação via Google Identity ou Microsoft Entra ID

Fluxo de Controle de Acesso:
1. **Acesso Super Admin:**
   - Login direto com suporte@caracore.com.br + senha
   - Acesso imediato às páginas administrativas
   - Não passa por fluxo OAuth externo

2. **Configuração Inicial do Sistema:**
   - Super admin configura parâmetros iniciais do sistema
   - Define políticas de aprovação por provedor
   - Configura notificações e alertas

3. **Solicitação de Acesso por Usuários OAuth:**
   - Usuários autenticados via Google/Microsoft podem solicitar acesso
   - Sistema coleta: nome, email, justificativa, provedor de autenticação
   - Solicitação fica pendente até aprovação do super admin
   - Usuário recebe feedback sobre status da solicitação

4. **Aprovação pelo Super Administrador:**
   - Super admin acessa approval-requests.html para ver solicitações pendentes
   - Pode aprovar, negar ou solicitar mais informações
   - Sistema mantém log de todas as decisões de aprovação
   - Usuários aprovados ganham acesso automático à Área 51

5. **Gerenciamento Contínuo:**
   - admin-users.html: visualizar, editar, remover usuários autorizados de ambos provedores

super-admin-setup.js (Exemplo atualizado)
// Sistema de autenticação híbrido para super administrador
class SuperAdminAuth {
  constructor() {
    this.backendUrl = 'https://caracore-backend.azurewebsites.net';
    this.superAdminEmail = 'suporte@caracore.com.br';
  }

  // Autenticação do super admin (independente de OAuth)
  async authenticateSuperAdmin(email, password) {
    if (email !== this.superAdminEmail) {
      throw new Error('E-mail não autorizado para super admin');
    }

    try {
      const response = await fetch(`${this.backendUrl}/auth/super-admin`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          email: email,
          password: password
        })
      });

      if (!response.ok) {
        throw new Error('Credenciais inválidas');
      }

      const result = await response.json();
      
      // Armazenar token de super admin
      localStorage.setItem('super_admin_token', result.token);
      localStorage.setItem('super_admin_email', email);
      localStorage.setItem('super_admin_authenticated', 'true');
      
      return result;
    } catch (error) {
      console.error('Erro na autenticação do super admin:', error);
      throw error;
    }
  }

  // Verificar se super admin está autenticado
  isSuperAdminAuthenticated() {
    const token = localStorage.getItem('super_admin_token');
    const authenticated = localStorage.getItem('super_admin_authenticated');
    return token && authenticated === 'true';
  }

  // Logout do super admin
  logoutSuperAdmin() {
    localStorage.removeItem('super_admin_token');
    localStorage.removeItem('super_admin_email');
    localStorage.removeItem('super_admin_authenticated');
    window.location.href = '/secure/index.html';
  }
}

// Configuração inicial do super administrador (primeira vez)
function setupSuperAdminInterface() {
  const loginForm = document.getElementById('super-admin-login-form');
  
  if (loginForm) {
    loginForm.addEventListener('submit', async (e) => {
      e.preventDefault();
      
      const email = document.getElementById('admin-email').value;
      const password = document.getElementById('admin-password').value;
      
      const superAuth = new SuperAdminAuth();
      
      try {
        await superAuth.authenticateSuperAdmin(email, password);
        
        // Redirecionar para painel administrativo
        window.location.href = '/secure/approval-requests.html';
        
      } catch (error) {
        showError('Erro na autenticação: ' + error.message);
      }
    });
  }
}

@app.route('/auth/super-admin', methods=['POST'])
def authenticate_super_admin():
    """Autenticação do super administrador"""
    data = request.get_json()
    email = data.get('email')
    password = data.get('password')
    
    # Verificar credenciais do super admin
    SUPER_ADMIN_EMAIL = 'suporte@caracore.com.br'
    SUPER_ADMIN_PASSWORD_HASH = os.getenv('SUPER_ADMIN_PASSWORD_HASH')
    
    if email != SUPER_ADMIN_EMAIL:
        return jsonify({'error': 'Email não autorizado'}), 401
    
    # Verificar senha hasheada
    import hashlib
    password_hash = hashlib.sha256(password.encode()).hexdigest()
    
    if password_hash != SUPER_ADMIN_PASSWORD_HASH:
        return jsonify({'error': 'Senha incorreta'}), 401
    
    # Gerar token JWT para super admin
    import jwt
    from datetime import datetime, timedelta
    
    payload = {
        'email': email,
        'role': 'super_admin',
        'exp': datetime.utcnow() + timedelta(hours=24)
    }
    
    token = jwt.encode(payload, os.getenv('JWT_SECRET_KEY'), algorithm='HS256')
    
    return jsonify({
        'token': token,
        'email': email,
        'role': 'super_admin',
        'expires_in': 86400  # 24 horas
    })

@app.route('/auth/verify-super-admin', methods=['POST'])
def verify_super_admin():
    """Verificar token do super administrador"""
    auth_header = request.headers.get('Authorization')
    
    if not auth_header or not auth_header.startswith('Bearer '):
        return jsonify({'error': 'Token não fornecido'}), 401
    
    token = auth_header.split(' ')[1]
    
    try:
        payload = jwt.decode(token, os.getenv('JWT_SECRET_KEY'), algorithms=['HS256'])
        
        if payload.get('role') != 'super_admin':
            return jsonify({'error': 'Token não é de super admin'}), 403
            
        return jsonify({
            'valid': True,
            'email': payload.get('email'),
            'role': payload.get('role')
        })
        
    except jwt.ExpiredSignatureError:
        return jsonify({'error': 'Token expirado'}), 401
    except jwt.InvalidTokenError:
        return jsonify({'error': 'Token inválido'}), 401

// Gerenciar aprovações de solicitações
function approvePendingRequest(requestId, provider) {
  const pendingRequests = JSON.parse(localStorage.getItem(`${provider}_pending_requests`) || '[]');
  const request = pendingRequests.find(req => req.id === requestId);
  
  if (request) {
    // Adicionar à lista de usuários autorizados
    const authorizedUsers = JSON.parse(localStorage.getItem(`${provider}_authorized_users`) || '[]');
    authorizedUsers.push({
      email: request.email,
      name: request.name,
      role: 'user',
      approvedAt: new Date().toISOString(),
      approvedBy: getCurrentSuperAdmin(provider)
    });
    localStorage.setItem(`${provider}_authorized_users`, JSON.stringify(authorizedUsers));
    
    // Remover da lista de pendentes
    const updatedPending = pendingRequests.filter(req => req.id !== requestId);
    localStorage.setItem(`${provider}_pending_requests`, JSON.stringify(updatedPending));
    
    // Log da ação
    logAdminAction('approve_request', { requestId, email: request.email, provider });
  }
}
